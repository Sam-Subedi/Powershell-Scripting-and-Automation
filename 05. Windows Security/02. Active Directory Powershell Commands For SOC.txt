# SOC Analyst - Windows & Active Directory Useful Commands
# Purpose: Practical commands for SOC investigation, identity and privilege analysis,
# account/group/computer investigation, process/service analysis, networking,
# persistence hunting, and Windows event log review.
#
# Notes:
# - Run commands from an elevated PowerShell session when required.
# - Active Directory commands require the ActiveDirectory PowerShell module.
# - Replace placeholders such as USERNAME, COMPUTERNAME, SERVER01, and DOMAIN.
# - A command finding something suspicious does NOT by itself prove malicious activity.
#   Always investigate the surrounding context, timeline, user, process, and events.


# ============================================================
# 1. CURRENT USER / HOST / DOMAIN INFORMATION
# ============================================================

whoami
Shows the currently logged-on user.

whoami /user
Shows the current user's username and SID.

whoami /groups
Shows groups the current user belongs to.

whoami /priv
Shows privileges assigned to the current user token.

whoami /all
Shows user, groups, privileges, SID, and other security information.

$env:USERNAME
Shows the current username.

$env:USERDOMAIN
Shows the user's domain.

$env:COMPUTERNAME
Shows the hostname.

hostname
Shows the computer hostname.

Get-CimInstance Win32_ComputerSystem |
Select-Object Name, Domain, PartOfDomain
Shows computer name, domain/workgroup, and whether the computer is domain joined.

Get-ComputerInfo
Shows detailed Windows computer information.

systeminfo
Shows detailed operating system, hardware, domain, and patch information.

net config workstation
Shows workstation configuration, including domain/workgroup information.


# ============================================================
# 2. LOCAL USER ACCOUNT INVESTIGATION
# ============================================================

net user
Lists local user accounts.

net user USERNAME
Shows detailed information about a specific local user.

net user Administrator
Shows details about the built-in Administrator account.

Get-LocalUser
Lists local user accounts.

Get-LocalUser -Name "Administrator"
Checks a specific local account.

Get-LocalUser |
Select-Object Name, Enabled, LastLogon
Shows useful account details.

Get-LocalUser |
Select-Object Name, Enabled, Description, LastLogon,
PasswordRequired, PasswordExpires, UserMayChangePassword
Shows detailed local account security properties.

Get-CimInstance Win32_UserAccount |
Select-Object Name, Domain, LocalAccount, Disabled, Lockout, PasswordRequired, SID
Shows local and domain account information.

Get-CimInstance Win32_UserAccount -Filter "LocalAccount = True"
Shows local user accounts only.

Get-CimInstance Win32_UserAccount -Filter "LocalAccount = True AND Disabled = False"
Shows enabled local user accounts.

Get-CimInstance Win32_UserAccount -Filter "LocalAccount = True AND PasswordRequired = False"
Identifies local accounts that do not require a password.

Search-ADAccount -LockedOut
Finds locked-out AD accounts.

Search-ADAccount -AccountDisabled
Finds disabled AD accounts.

Search-ADAccount -PasswordNeverExpires
Finds AD accounts configured with passwords that never expire.

# 10A. WINDOWS SERVICE MANAGEMENT / CONTAINMENT
# ============================================================

Get-Service -Name "SERVICE_NAME"
Checks a specific Windows service.

Start-Service -Name "SERVICE_NAME"
Starts a Windows service.

Stop-Service -Name "SERVICE_NAME"
Stops a Windows service.

Restart-Service -Name "SERVICE_NAME"
Restarts a Windows service.

Set-Service -Name "SERVICE_NAME" -StartupType Disabled
Changes a service startup type to Disabled.

Set-Service -Name "SERVICE_NAME" -StartupType Automatic
Changes a service startup type to Automatic.

Get-CimInstance Win32_Service |
Select-Object Name,State,StartMode,StartName,PathName
Reviews service state, startup mode, service account, and executable path.

# SOC/security use:
# Stop-Service may be used as a containment action when an approved response
# procedure requires stopping a suspicious service.
# Do not stop critical production services without authorization.

# 9A. PROCESS MANAGEMENT / CONTAINMENT
# ============================================================

Get-Process -Id PROCESSID
Gets information about a process by PID.

Stop-Process -Id PROCESSID
Stops a process by process ID.

Stop-Process -Name "PROCESSNAME"
Stops processes matching the specified process name.

Stop-Process -Id PROCESSID -Force
Forcefully terminates a process when required.

Start-Process "C:\Path\program.exe"
Starts a process/application.

Get-CimInstance Win32_Process |
Select-Object Name,ProcessId,ParentProcessId,ExecutablePath,CommandLine
Reviews process execution and lineage.

# SOC/security use:
# Stopping a process can be an incident-containment action, but first capture
# relevant evidence such as PID, command line, executable path, hash, parent
# process, user, and network connections when the investigation requires it.

# 9. PROCESS INVESTIGATION
# ============================================================

Get-Process
Lists running processes.

Get-Process | Sort-Object CPU -Descending
Shows processes using the most CPU.

Get-Process | Sort-Object WorkingSet -Descending
Shows processes using the most memory.

Get-Process powershell
Checks for running PowerShell processes.

Get-Process cmd
Checks for running Command Prompt processes.

Get-Process -Id PROCESSID
Gets information about a specific process.

Get-CimInstance Win32_Process
Provides detailed process information.

Get-CimInstance Win32_Process |
Select-Object Name, ProcessId, ParentProcessId, ExecutablePath, CommandLine
Shows process name, PID, parent PID, executable path, and command line.

Get-CimInstance Win32_Process |
Where-Object { $_.ExecutablePath -like '*\AppData\*' } |
Select-Object Name, ProcessId, ParentProcessId, ExecutablePath, CommandLine
Finds processes executing from AppData and displays useful process lineage information.

Get-CimInstance Win32_Process |
Where-Object { $_.Name -match 'powershell.exe|cmd.exe|wscript.exe|cscript.exe|mshta.exe|rundll32.exe|regsvr32.exe' } |
Select-Object Name, ProcessId, ParentProcessId, ExecutablePath, CommandLine
Looks for commonly investigated Windows execution processes.

Get-Process -Id PARENT_PROCESS_ID
Retrieves information about a process using its PID.

# SOC investigation focus:
# Process -> Parent Process -> Command Line -> File Path -> User -> Network Connection
